    function validateUniqueAgentNumbers() {
        var numbers = {};
        var duplicateFound = false;

        for (let i = 1; i <= 4; i++) {
            var numEl = document.getElementById("Dynamic_agtNumber" + i);
            if (!numEl || !numEl.value.trim()) continue;

            var num = numEl.value.trim();
            if (numbers[num]) {
                duplicateFound = true;
                alert("Error, Do not add same Agent ID on multiple rows. Agent ID must be unique for percentages. Agent Number " + num + " has been entered more than once.");
                numEl.value = "";
                if (numEl.id === "Dynamic_agtNumber1") {
                    document.getElementById("Dynamic_agtPercent1").value = "";
                }
                break;
            }
            numbers[num] = true;
        }

        return !duplicateFound;
    }
