        function checkAudioDurationFromAPI() {
            var AppNum = document.getElementById('appNum').value;
            var agentID = document.getElementById('hdnAgentID').value;

            var apiUrl = "https://web2.aatx.com/api/aatxapi/api/AudioFileSpecs/" + AppNum + "/" + agentID;

           var xhttp = new XMLHttpRequest();
           xhttp.onreadystatechange = function () {
               if (this.readyState === 4) {
                   if (this.status === 200) {
                       try {
                           var response = JSON.parse(this.responseText);

                           var duration = response.duration_in_seconds || 0;
                           if (duration < 180) {
                               alert(
                                   "Please verify that you have uploaded a voice signature that is read from the script provided.\n\n" +
                                   "Uploading an incorrect voice signature could cause delays in processing the application.\n\n" +
                                   "Recording is less than 3 minutes.\n Duration: " + duration.toFixed(1) + " seconds."
                               );
                           } 
                       } catch (err) {
                           alert("Error reading API response.");
                       }
                   } else {
                       alert("Request failed. Status: " + this.status);
                   }
               }
           };

           xhttp.open("GET", apiUrl, true);
           xhttp.send();
       }
