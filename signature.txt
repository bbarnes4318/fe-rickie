    function validateMentalHealthSelection() {
        var container = document.getElementById('QuestMentalInfo');

        
        if (!container || window.getComputedStyle(container).display === 'none') {
            
            return true; // Skip validation if hidden
        }

        var checkboxes = container.querySelectorAll("input[type='checkbox']");
        var isChecked = Array.from(checkboxes).some(checkbox => checkbox.checked);

        if (!isChecked) {
             setError('Please select at least one condition in the Mental Health Questionnaire.');
            return false; 
        }
          var checkBoxOther = document.getElementById('MHQConditionsOther');
          var textAreaOther = document.getElementById('OtherComments');

        if (checkBoxOther && checkBoxOther.checked && textAreaOther && textAreaOther.value.trim() === '') {
             setError('Please provide details in the Other comments section.');
            textAreaOther.focus();
            return false; 
        }
        return true;
    }
